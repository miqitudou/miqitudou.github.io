窗口函数实现序号递增

DENSE_RANK() OVER (ORDER BY tm.meet_number) AS '序号' # 相同的会议编号归为相同序号

ROW_NUMBER() OVER (PARTITION BY tm.meet_number, ud.id ORDER BY mp.id) AS 截图序号 #以会议编号+医生ID分组，在组内以mp的id排序
